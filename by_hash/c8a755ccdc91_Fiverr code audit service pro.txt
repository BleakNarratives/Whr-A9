#!/usr/bin/env python3
"""
EquiNex Pro - Professional Code Audit Report Generator
Save as: generate_report.py
"""

import json
import os
from datetime import datetime
from fpdf import FPDF
import matplotlib.pyplot as plt
import numpy as np
from pathlib import Path

class AuditReportGenerator:
    def __init__(self, analysis_data):
        self.data = analysis_data
        self.report_date = datetime.now().strftime("%B %d, %Y")
        
    def generate_complete_report(self):
        """Generate a professional PDF report"""
        pdf = FPDF()
        
        # Setup
        pdf.set_auto_page_break(auto=True, margin=15)
        
        # Cover Page
        self._create_cover_page(pdf)
        
        # Executive Summary
        self._create_executive_summary(pdf)
        
        # Technical Analysis
        self._create_technical_analysis(pdf)
        
        # Security Assessment
        self._create_security_assessment(pdf)
        
        # Performance Analysis
        self._create_performance_analysis(pdf)
        
        # Recommendations
        self._create_recommendations(pdf)
        
        # Action Plan
        self._create_action_plan(pdf)
        
        # Save report
        filename = f"Code_Audit_Report_{datetime.now().strftime('%Y%m%d_%H%M')}.pdf"
        pdf.output(filename)
        return filename
    
    def _create_cover_page(self, pdf):
        pdf.add_page()
        pdf.set_fill_color(15, 23, 42)  # Dark blue
        pdf.rect(0, 0, 210, 297, 'F')
        
        # Title
        pdf.set_font('Arial', 'B', 24)
        pdf.set_text_color(255, 255, 255)
        pdf.cell(0, 40, '', 0, 1, 'C')
        pdf.cell(0, 20, '🔍 PROFESSIONAL CODE AUDIT', 0, 1, 'C')
        
        # Subtitle
        pdf.set_font('Arial', 'I', 16)
        pdf.set_text_color(101, 163, 13)  # Green
        pdf.cell(0, 10, 'Technical Debt Assessment & Security Analysis', 0, 1, 'C')
        
        # Client Info
        pdf.set_font('Arial', '', 12)
        pdf.set_text_color(200, 200, 200)
        pdf.cell(0, 60, '', 0, 1)
        pdf.cell(0, 8, f'Prepared for: {self.data.get("client", "Client")}', 0, 1, 'C')
        pdf.cell(0, 8, f'Date: {self.report_date}', 0, 1, 'C')
        pdf.cell(0, 8, 'EquiNex Pro Code Analytics', 0, 1, 'C')
        
        # Confidential
        pdf.set_font('Arial', 'I', 10)
        pdf.set_text_color(150, 150, 150)
        pdf.cell(0, 100, '', 0, 1)
        pdf.cell(0, 8, 'CONFIDENTIAL - FOR INTERNAL USE ONLY', 0, 1, 'C')
    
    def _create_executive_summary(self, pdf):
        pdf.add_page()
        pdf.set_fill_color(255, 255, 255)
        pdf.set_text_color(0, 0, 0)
        
        # Header
        pdf.set_font('Arial', 'B', 18)
        pdf.cell(0, 10, 'EXECUTIVE SUMMARY', 0, 1)
        pdf.ln(5)
        
        # Risk Score Box
        risk_score = self.data.get('debt_score', 65)
        risk_color = (220, 38, 38) if risk_score > 70 else (245, 158, 11) if risk_score > 40 else (34, 197, 94)
        
        pdf.set_fill_color(*risk_color)
        pdf.set_text_color(255, 255, 255)
        pdf.set_font('Arial', 'B', 14)
        pdf.cell(40, 12, f'RISK SCORE: {risk_score}/100', 1, 0, 'C', True)
        
        pdf.set_text_color(0, 0, 0)
        pdf.set_font('Arial', 'B', 10)
        pdf.cell(10)
        status = "HIGH RISK" if risk_score > 70 else "MEDIUM RISK" if risk_score > 40 else "LOW RISK"
        pdf.cell(0, 12, f'STATUS: {status}', 0, 1)
        pdf.ln(10)
        
        # Key Findings
        pdf.set_font('Arial', 'B', 14)
        pdf.cell(0, 10, 'Key Findings', 0, 1)
        pdf.set_font('Arial', '', 11)
        
        findings = [
            f"• {self.data.get('critical_issues', 0)} critical issues requiring immediate attention",
            f"• {self.data.get('security_risks', 0)} security vulnerabilities identified",
            f"• Code maintainability score: {self.data.get('maintainability', 0)}%",
            f"• Estimated refactoring effort: {self.data.get('fix_cost', 0)} developer hours",
            f"• Performance improvement potential: {self.data.get('performance_gain', 0)}%"
        ]
        
        for finding in findings:
            pdf.multi_cell(0, 6, finding)
            pdf.ln(2)
    
    def _create_technical_analysis(self, pdf):
        pdf.add_page()
        pdf.set_font('Arial', 'B', 18)
        pdf.cell(0, 10, 'TECHNICAL ANALYSIS', 0, 1)
        pdf.ln(5)
        
        # Code Quality Metrics
        pdf.set_font('Arial', 'B', 14)
        pdf.cell(0, 8, 'Code Quality Metrics', 0, 1)
        pdf.set_font('Arial', '', 11)
        
        metrics = self.data.get('metrics', {})
        for metric, value in metrics.items():
            pdf.cell(90, 6, f'• {metric.replace("_", " ").title()}:', 0, 0)
            pdf.cell(0, 6, str(value), 0, 1)
        
        pdf.ln(10)
        
        # File Analysis
        pdf.set_font('Arial', 'B', 14)
        pdf.cell(0, 8, 'File Analysis Summary', 0, 1)
        pdf.set_font('Arial', '', 11)
        
        files = self.data.get('files_analyzed', [])
        for file_info in files[:10]:  # Show top 10
            pdf.cell(0, 6, f'• {file_info["name"]} - Complexity: {file_info.get("complexity", "N/A")}/10', 0, 1)
    
    def _create_security_assessment(self, pdf):
        pdf.add_page()
        pdf.set_font('Arial', 'B', 18)
        pdf.cell(0, 10, 'SECURITY ASSESSMENT', 0, 1)
        pdf.ln(5)
        
        # Security Findings
        security_issues = self.data.get('security_issues', [])
        
        for i, issue in enumerate(security_issues, 1):
            # Risk indicator
            risk_level = issue.get('risk', 'medium')
            risk_color = {'high': (220, 38, 38), 'medium': (245, 158, 11), 'low': (34, 197, 94)}
            
            pdf.set_fill_color(*risk_color.get(risk_level, (245, 158, 11)))
            pdf.set_text_color(255, 255, 255)
            pdf.set_font('Arial', 'B', 10)
            pdf.cell(25, 8, risk_level.upper(), 1, 0, 'C', True)
            
            pdf.set_text_color(0, 0, 0)
            pdf.set_font('Arial', 'B', 12)
            pdf.cell(5)
            pdf.cell(0, 8, f'{i}. {issue["title"]}', 0, 1)
            
            pdf.set_font('Arial', '', 10)
            pdf.multi_cell(0, 5, f'Location: {issue.get("file", "Unknown")}')
            pdf.multi_cell(0, 5, f'Description: {issue.get("description", "")}')
            
            if issue.get('recommendation'):
                pdf.set_font('Arial', 'I', 10)
                pdf.multi_cell(0, 5, f'Recommendation: {issue["recommendation"]}')
            
            pdf.ln(5)
    
    def _create_performance_analysis(self, pdf):
        pdf.add_page()
        pdf.set_font('Arial', 'B', 18)
        pdf.cell(0, 10, 'PERFORMANCE ANALYSIS', 0, 1)
        pdf.ln(5)
        
        # Performance Issues
        performance_issues = self.data.get('performance_issues', [])
        
        for issue in performance_issues:
            pdf.set_font('Arial', 'B', 12)
            pdf.cell(0, 8, f'• {issue["title"]}', 0, 1)

            pdf.multi_cell(0, 5, f'Potential Improvement: {issue.get("improvement", "")}')
            pdf.ln(3)
    
    def _create_recommendations(self, pdf):
        pdf.add_page()
        pdf.set_font('Arial', 'B', 18)
        pdf.cell(0, 10, 'RECOMMENDATIONS', 0, 1)

            pdf.cell(35, 8, f'PRIORITY: {priority.upper()}', 1, 0, 'C', True)
            
            pdf.set_text_color(0, 0, 0)
            pdf.set_font('Arial', 'B', 12)
            pdf.cell(5)
            pdf.cell(0, 8, f'{i}. {rec["title"]}', 0, 1)
            
            pdf.set_font('Arial', '', 10)
            pdf.multi_cell(0, 5, f'Description: {rec.get("description", "")}')

    def safe_text(self, text):
        """Ensure text is Latin-1 compatible"""
        if text is None:
            return ""
        # Replace problematic Unicode characters
        replacements = {
            '•': '-', '→': '->', '⚠️': '[!]', '🔍': '[ANALYZE]',
            '🚨': '[ALERT]', '💡': '[IDEA]', '🎯': '[TARGET]',
            '📅': '[CALENDAR]', '📊': '[CHART]', '🔒': '[LOCK]'
        }
        safe_text = str(text)
        for char, replacement in replacements.items():
            safe_text = safe_text.replace(char, replacement)
        return safe_text.encode('latin-1', 'replace').decode('latin-1')

#!/usr/bin/env python3
"""
EquiNex Pro - Safe GitHub Repository Analyzer
"""

import os
import tempfile
import subprocess
import shutil
from audit_engine import analyze_codebase

def analyze_github_repo(repo_url, client_name="GitHub Client"):
    """Safely analyze a GitHub repository"""
    temp_dir = None
    try:
        # Create temporary directory
        temp_dir = tempfile.mkdtemp(prefix="equinex_audit_")
        print(f"📁 Created temporary directory: {temp_dir}")
        
        # Validate URL
        if not repo_url.startswith(('https://github.com/', 'https://gitlab.com/')):
            raise ValueError("Only GitHub and GitLab URLs are supported for security")
        
        # Clone the repo (read-only, depth=1 for speed)
        print(f"📥 Cloning repository: {repo_url}")
        result = subprocess.run([
            'git', 'clone', '--depth', '1', '--single-branch', repo_url, temp_dir
        ], capture_output=True, text=True, timeout=300)  # 5 minute timeout
        
        if result.returncode != 0:
            error_msg = result.stderr or "Unknown git error"
            # Provide helpful error messages
            if "fatal: repository" in error_msg and "not found" in error_msg:
                raise ValueError("Repository not found or not accessible")
            elif "fatal: could not read Username" in error_msg:
                raise ValueError("Repository may be private - only public repos supported")
            else:
                raise Exception(f"Git clone failed: {error_msg}")
        
        # Analyze the code
        print("🔍 Analyzing codebase...")
        analysis = analyze_codebase(temp_dir)
        analysis['client'] = client_name
        analysis['repo_url'] = repo_url
        
        print("✅ Analysis complete!")
        return analysis
        
    except subprocess.TimeoutExpired:
        raise Exception("Repository cloning timed out - repository may be too large")
    except Exception as e:
        raise e
    finally:
        # Always clean up, even on error
        if temp_dir and os.path.exists(temp_dir):
            print("🧹 Cleaning up temporary files...")
            shutil.rmtree(temp_dir, ignore_errors=True)
