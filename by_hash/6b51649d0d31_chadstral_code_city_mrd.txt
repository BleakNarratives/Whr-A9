#### **B. Camera Swings (Left/Right Rotation)**
Add this to your `gameLoop()` in the `<script>` section:
```javascript
// Camera swing variables
let cameraAngle = 0;
let swingDirection = 1; // 1 for right, -1 for left
let swingSpeed = 0.005;

// Inside gameLoop(), add before renderer.render():
cameraAngle += swingDirection * swingSpeed;
if (cameraAngle > 0.5) swingDirection = -1;
if (cameraAngle < -0.5) swingDirection = 1;
camera.position.x = 50 + Math.sin(cameraAngle) * 30;
camera.position.z = 100 + Math.cos(cameraAngle) * 30;
camera.lookAt(0, 0, 0);
```

#### **C. Storefront Debug View**
Add this function and bind it to a key (e.g., press `D`):
```javascript
// Storefront mode
let storefrontMode = false;
function toggleStorefront() {
  storefrontMode = !storefrontMode;
  if (storefrontMode) {
    camera.position.set(0, 20, 50); // Focus on a building
    camera.lookAt(buildings[0].position.x, 0, buildings[0].position.z);
    document.getElementById('codeEditor').style.display = 'none';
    showBuildingStats(buildings[0]);
  } else {
    camera.position.set(50, 80, 100); // Default view
    camera.lookAt(0, 0, 0);
    document.getElementById('codeEditor').style.display = 'block';
  }
}

function showBuildingStats(building) {
  const stats = document.createElement('div');
  stats.style.position = 'absolute';
  stats.style.top = '50%';
  stats.style.left = '50%';
  stats.style.transform = 'translate(-50%, -50%)';
  stats.style.background = 'rgba(0, 0, 0, 0.8)';
  stats.style.padding = '20px';
  stats.style.border = '2px solid #00ff44';
  stats.style.zIndex = '300';
  stats.style.color = '#00ff44';
  stats.style.fontFamily = 'Courier New, monospace';
  stats.innerHTML = `
    <h2>🏢 BUILDING STATS</h2>
    <p><strong>Type:</strong> ${building.userData.type}</p>
    <p><strong>Health:</strong> ${building.userData.health}/${building.userData.maxHealth}</p>
    <p><strong>Reward:</strong> $${building.userData.reward}</p>
    <p><strong>Bugs Nearby:</strong> ${bugs.filter(b => b.userData.targetBuilding === building).length}</p>
    <button onclick="enterBuilding(this)" data-building='${buildings.indexOf(building)}'>ENTER</button>
    <button onclick="document.body.removeChild(this.parentElement)">EXIT</button>
  `;
  document.body.appendChild(stats);
}
```

- Agents deployed in-game can trigger real Termux scripts (e.g., notifications, backups).
- Example: `termux_automations/deploy_swarm.sh` runs when you deploy an agent.
