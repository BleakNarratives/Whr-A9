## BOARDROOM-SPECIFIC VOICES (Zero Budget Edition)

**The Challenge:**
- Custom voice synthesis costs $$$ (ElevenLabs, etc.)
- Free TTS (gTTS) only has generic voices
- You want CFO to sound different from CTO, etc.

**The Solution: PROSODY HACKING**

We can't change the voice actor, but we can change:
- **Speed** (CFO = slower, deliberate; CTO = faster, excited)
- **Pitch** (shift up/down slightly)
- **Pauses** (add dramatic pauses for emphasis)
- **Word choice** (each persona has distinct vocabulary)

### Updated Audio Generator with Persona Voices

```python
#!/usr/bin/env python3
"""
Boardroom Audio Overview
Each director has distinct speaking style + voice modulation
"""

from gtts import gTTS
from pydub import AudioSegment
from pydub.playback import play
import re

class BoardroomAudioGenerator:
    def __init__(self):
        # Persona voice profiles (using gTTS variants + modulation)
        self.personas = {
            "CEO": {
                "voice": {"lang": "en", "tld": "com"},  # US English
                "speed": 1.0,  # Normal pace
                "pitch_shift": 0,  # No pitch change
                "style": "authoritative",
                "filler_words": ["clearly", "fundamentally", "strategically"],
            },
            "CFO": {
                "voice": {"lang": "en", "tld": "co.uk"},  # UK English
                "speed": 0.85,  # Slower, more deliberate
                "pitch_shift": -50,  # Slightly lower pitch (authoritative)
                "style": "analytical",
                "filler_words": ["frankly", "the numbers show", "bottom line"],
            },
            "CTO": {
                "voice": {"lang": "en", "tld": "com.au"},  # Australian English
                "speed": 1.15,  # Faster (excited about tech)
                "pitch_shift": 50,  # Slightly higher pitch (energetic)
                "style": "technical",
                "filler_words": ["technically", "actually", "basically"],
            },
            "CMO": {
                "voice": {"lang": "en", "tld": "ca"},  # Canadian English
                "speed": 1.05,  # Slightly faster (enthusiastic)
                "pitch_shift": 100,  # Higher pitch
                "style": "enthusiastic",
                "filler_words": ["honestly", "I think", "you know"],
            },
            "COO": {
                "voice": {"lang": "en", "tld": "co.in"},  # Indian English
                "speed": 0.95,  # Measured pace
                "pitch_shift": -25,  # Slightly lower
                "style": "pragmatic",
                "filler_words": ["operationally", "from my perspective", "realistically"],
            },
        }
    
    def speak(self, persona: str, text: str) -> AudioSegment:
        """Generate speech for specific persona with voice modulation"""
        
        if persona not in self.personas:
            persona = "CEO"  # Default
        
        profile = self.personas[persona]
        
        # Add persona-specific filler words occasionally (natural speech)
        text = self._add_personality(text, profile["filler_words"])
        
        # Generate base audio
        tts = gTTS(text=text, **profile["voice"])
        temp_file = f"/tmp/tts_{persona}.mp3"
        tts.save(temp_file)
        
        # Load and modulate
        audio = AudioSegment.from_mp3(temp_file)
        
        # Apply speed change
        if profile["speed"] != 1.0:
            audio = self._change_speed(audio, profile["speed"])
        
        # Apply pitch shift
        if profile["pitch_shift"] != 0:
            audio = self._change_pitch(audio, profile["pitch_shift"])
        
        return audio
    
    def _add_personality(self, text: str, fillers: list) -> str:
        """Inject filler words to match persona style"""
        # 20% chance to add a filler at the start
        import random
        if random.random() < 0.2:
            filler = random.choice(fillers)
            text = f"{filler}, {text}"
        return text
    
    def _change_speed(self, audio: AudioSegment, speed: float) -> AudioSegment:
        """Change playback speed without pitch shift (closest approximation)"""
        # This is a hack - pydub doesn't have true time-stretching
        # We change frame rate which affects both speed and pitch
        # Then we'll correct pitch separately
        
        sound_with_altered_frame_rate = audio._spawn(
            audio.raw_data, 
            overrides={"frame_rate": int(audio.frame_rate * speed)}
        )
        
        # Convert back to original frame rate (keeps speed change)
        return sound_with_altered_frame_rate.set_frame_rate(audio.frame_rate)
    
    def _change_pitch(self, audio: AudioSegment, semitones: int) -> AudioSegment:
        """Shift pitch up or down"""
        # Pitch shift formula: multiply sample rate
        new_sample_rate = int(audio.frame_rate * (2.0 ** (semitones / 12.0)))
        
        pitched_sound = audio._spawn(
            audio.raw_data,
            overrides={'frame_rate': new_sample_rate}
        )
        
        # Return to original sample rate (keeps pitch change)
        return pitched_sound.set_frame_rate(audio.frame_rate)
    
    def generate_boardroom_discussion(self, script: dict, output_file: str = "boardroom_audio.mp3"):
        """Create audio from boardroom script with persona voices"""
        
        audio_segments = []
        
        # Intro
        if "intro" in script:
            for persona, text in script["intro"].items():
                audio_segments.append(self.speak(persona, text))
                audio_segments.append(AudioSegment.silent(duration=500))
        
        # Main discussion segments
        for segment in script.get("segments", []):
            # Topic announcement (brief pause before topic shift)
            audio_segments.append(AudioSegment.silent(duration=1000))
            
            for line in segment.get("dialogue", []):
                persona = line["speaker"]
                text = line["text"]
                
                audio_segments.append(self.speak(persona, text))
                audio_segments.append(AudioSegment.silent(duration=400))
        
        # Outro
        if "outro" in script:
            audio_segments.append(AudioSegment.silent(duration=800))
            for persona, text in script["outro"].items():
                audio_segments.append(self.speak(persona, text))
                audio_segments.append(AudioSegment.silent(duration=300))
        
        # Combine all
        final_audio = sum(audio_segments)
        
        # Export
        final_audio.export(output_file, format="mp3", bitrate="128k")
        print(f"✓ Boardroom discussion saved to {output_file}")
        
        return output_file
```

from firefly.firewarden import FireWarden
import matplotlib.pyplot as plt
import pandas as pd

class FireflyAnalytics:
    def __init__(self):
        self.scouts = []
        self.warden = FireWarden()
        self.data_streams = {}
    
    def add_data_source(self, source_type: str, config: dict):
        """Add data source (API, web scraper, CSV, etc.)"""
        scout = Scout(source_type, config)
        self.scouts.append(scout)
        scout.start_monitoring()
    
    def generate_dashboard(self, metrics: list, output: str = "dashboard.html"):
        """Create interactive HTML dashboard"""
        
        data = self.warden.aggregate_data(self.scouts)
        
        # Generate charts
        fig, axes = plt.subplots(len(metrics), 1, figsize=(12, 4*len(metrics)))
        
        for i, metric in enumerate(metrics):
            df = pd.DataFrame(data[metric])
            df.plot(ax=axes[i], title=metric)
        
        plt.tight_layout()
        plt.savefig("/tmp/charts.png")
        
        # Embed in HTML
        html = f"""
        <html>
        <head><title>Firefly Analytics</title></head>
        <body>
            <h1>Real-Time Dashboard</h1>
            <img src="/tmp/charts.png" />
            <div id="live-data"></div>
            <script>
                // WebSocket for live updates
                setInterval(() => {{
                    fetch('/api/live_data')
                        .then(r => r.json())
                        .then(d => updateCharts(d));
                }}, 5000);
            </script>
        </body>
        </html>
        """
        
        with open(output, 'w') as f:
            f.write(html)

#!/usr/bin/env python3
"""
Claude DAW - Rapper Module
AI-assisted music production for broke artists
"""

import google.generativeai as genai
from mido import MidiFile, MidiTrack, Message
from pydub import AudioSegment

class RapperModule:
    def __init__(self, api_key: str):
        genai.configure(api_key=api_key)
        self.model = genai.GenerativeModel('gemini-2.0-flash-exp')
    
    def generate_lyrics(self, topic: str, style: str = "boom bap", bars: int = 16):
        """Generate rap lyrics"""
        
        prompt = f"""Write {bars} bars of rap lyrics about: {topic}

Style: {style}
Requirements:
- Tight rhyme scheme (AABB or ABAB)
- Internal rhymes
- Wordplay/metaphors
- Flow that works over {style} beat (adjust syllable count)

Format as:
[Bar 1] lyrics
[Bar 2] lyrics
..."""

        response = self.model.generate_content(prompt)
        return response.text
    
    def create_beat(self, style: str = "trap", bpm: int = 140, bars: int = 16):
        """Generate MIDI beat pattern"""
        
        mid = MidiFile()
        track = MidiTrack()
        mid.tracks.append(track)
        
        # Set tempo
        track.append(Message('set_tempo', tempo=int(60000000 / bpm)))
        
        # Generate drum pattern (kick, snare, hi-hat)
        if style == "trap":
            # Trap pattern: kick on 1, snare on 3, rapid hi-hats
            pattern = self._trap_pattern(bars)
        elif style == "boom bap":
            # Boom bap: heavy kick/snare, laid-back hats
            pattern = self._boom_bap_pattern(bars)
        
        for note, velocity, time in pattern:
            track.append(Message('note_on', note=note, velocity=velocity, time=time))
            track.append(Message('note_off', note=note, velocity=0, time=96))
        
        mid.save('/tmp/beat.mid')
        print("Beat saved: /tmp/beat.mid")
        return mid
    
    def _trap_pattern(self, bars: int):
        # Simplified trap drum pattern
        pattern = []
        ticks_per_bar = 384
        
        for bar in range(bars):
            offset = bar * ticks_per_bar
            
            # Kick (808) on 1 and 3.5
            pattern.append((36, 100, offset))
            pattern.append((36, 100, offset + 336))
            
            # Snare on 2 and 4
            pattern.append((38, 90, offset + 192))
            pattern.append((38, 90, offset + 576))
            
            # Hi-hats (rapid 16ths)
            for i in range(16):
                pattern.append((42, 60, offset + i*24))
        
        return pattern
    
    def _boom_bap_pattern(self, bars: int):
        # Classic boom bap
        pattern = []
        ticks_per_bar = 384
        
        for bar in range(bars):
            offset = bar * ticks_per_bar
            
            # Heavy kick on 1 and 3
            pattern.append((36, 110, offset))
            pattern.append((36, 110, offset + 384))
            
            # Snare on 2 and 4
            pattern.append((38, 100, offset + 192))
            pattern.append((38, 100, offset + 576))
            
            # Laid back hi-hats (8ths)
            for i in range(8):
                pattern.append((42, 70, offset + i*48))
        
        return pattern
    
    def analyze_flow(self, lyrics: str, bpm: int = 140):
        """Analyze if flow matches beat"""
        
        prompt = f"""Analyze this rap flow for a {bpm} BPM beat:

{lyrics}

Check:
- Syllable count per bar (should be ~16-20 for typical flow)
- Rhyme scheme consistency
- Awkward phrasing/tongue twisters
- Breath control (too many syllables without pause?)

Suggest improvements."""

        response = self.model.generate_content(prompt)
        return response.text

#!/usr/bin/env python3
"""
File Handler Utilities
Document loading, parsing, and management
"""

from pathlib import Path
import json
import yaml

class FileHandler:
    """Universal file I/O utilities"""
    
    @staticmethod
    def load_document(filepath: str) -> str:
        """Load text from various file types"""
        path = Path(filepath)
        
        if path.suffix in ['.txt', '.md']:
            return path.read_text()
        elif path.suffix == '.json':
            return json.dumps(json.loads(path.read_text()), indent=2)
        elif path.suffix in ['.yaml', '.yml']:
            return yaml.dump(yaml.safe_load(path.read_text()))
        # TODO: Add PDF, DOCX support
        
        return path.read_text()
    
    @staticmethod
    def save_output(content: str, filepath: str, format: str = 'auto'):
        """Save content with automatic format detection"""
        path = Path(filepath)
        path.parent.mkdir(exist_ok=True, parents=True)
        
        if format == 'auto':
            format = path.suffix.lstrip('.')
        
        # TODO: Format-specific saving
        path.write_text(content)

#!/usr/bin/env python3
"""
Professional Audio Processor
Using industry-standard libraries
"""

import librosa
import soundfile as sf
from pedalboard import Pedalboard, Reverb, Compressor, HighpassFilter, LowpassFilter
import numpy as np

class ProAudioProcessor:
    def __init__(self, sample_rate=44100):
        self.sr = sample_rate
        self.board = Pedalboard()
    
    def load(self, filepath):
        """Load audio with librosa (handles any format)"""
        audio, sr = librosa.load(filepath, sr=self.sr, mono=False)
        return audio, sr
    
    def add_reverb(self, audio, room_size=0.5, damping=0.5):
        """Spotify-quality reverb"""
        reverb = Reverb(room_size=room_size, damping=damping)
        return reverb(audio, self.sr)
    
    def compress(self, audio, threshold_db=-20, ratio=4):
        """Professional compression"""
        comp = Compressor(threshold_db=threshold_db, ratio=ratio)
        return comp(audio, self.sr)
    
    def eq_3band(self, audio, low_gain=0, mid_gain=0, high_gain=0):
        """3-band EQ with filters"""
        board = Pedalboard([
            HighpassFilter(cutoff_frequency_hz=80),     # Cut sub-bass
            # TODO: Add parametric mid-band EQ
            LowpassFilter(cutoff_frequency_hz=16000),   # Cut harsh highs
        ])
        return board(audio, self.sr)
    
    def analyze_bpm(self, audio):
        """Detect tempo"""
        tempo, beats = librosa.beat.beat_track(y=audio, sr=self.sr)
        return tempo
    
    def separate_stems(self, filepath):
        """Isolate vocals, drums, bass, other (using Spleeter)"""
        from spleeter.separator import Separator
        separator = Separator('spleeter:4stems')
        separator.separate_to_file(filepath, '/tmp/')
        return '/tmp/vocals.wav', '/tmp/drums.wav', '/tmp/bass.wav', '/tmp/other.wav'
    
    def export(self, audio, filepath, format='wav'):
        """High-quality export"""
        sf.write(filepath, audio.T, self.sr, format=format)

#!/usr/bin/env python3
from mido import MidiFile, MidiTrack, Message

class BeatGenerator:
    def __init__(self, bpm=140):
        self.bpm = bpm
    
    def create_beat(self, style='trap', bars=16):
        mid = MidiFile()
        track = MidiTrack()
        mid.tracks.append(track)
        
        # Set tempo
        tempo = int(60000000 / self.bpm)
        track.append(Message('program_change', program=0, time=0))
        
        # Generate pattern based on style
        if style == 'trap':
            pattern = self._trap_pattern(bars)
        else:
            pattern = self._boom_bap_pattern(bars)
        
        for note, velocity, time in pattern:
            track.append(Message('note_on', channel=9, note=note, velocity=velocity, time=time))
            track.append(Message('note_off', channel=9, note=note, velocity=0, time=96))
        
        return mid
    
    def _trap_pattern(self, bars):
        pattern = []
        ticks = 384  # ticks per bar
        
        for bar in range(bars):
            offset = bar * ticks
            
            # Kick on 1 and 3.5
            pattern.append((36, 100, offset if bar == 0 else 0))
            pattern.append((36, 100, ticks // 3))
            
            # Snare on 2 and 4
            pattern.append((38, 90, ticks // 4))
            pattern.append((38, 90, ticks // 2))
            
            # Hi-hats (16th notes)
            for i in range(16):
                pattern.append((42, 60, ticks // 32))
        
        return pattern
    
    def _boom_bap_pattern(self, bars):
        # Simpler pattern
        pattern = []
        ticks = 384
        
        for bar in range(bars):
            offset = bar * ticks
            
            pattern.append((36, 110, offset if bar == 0 else 0))  # Kick on 1
            pattern.append((38, 100, ticks // 2))  # Snare on 2
            pattern.append((36, 110, ticks // 4))  # Kick on 3
            pattern.append((38, 100, ticks // 4))  # Snare on 4
        
        return pattern

if __name__ == '__main__':

    beat = gen.create_beat(style='trap', bars=8)
    beat.save('/tmp/test_beat.mid')

"""Beat Generator - MIDI beat creation"""
from mido import MidiFile, MidiTrack, Message

class BeatGenerator:
    def __init__(self, bpm=140):
        self.bpm = bpm
    
    def create_beat(self, style='trap', bars=16):
        mid = MidiFile()
        track = MidiTrack()
        mid.tracks.append(track)
        
        # Set tempo
        tempo = int(60000000 / self.bpm)
        track.append(Message('program_change', program=0, time=0))
        
        if style == 'trap':
            pattern = self._trap_pattern(bars)
        else:
            pattern = self._boom_bap_pattern(bars)
        
        for note, velocity, time in pattern:
            track.append(Message('note_on', channel=9, note=note, velocity=velocity, time=time))
            track.append(Message('note_off', channel=9, note=note, velocity=0, time=96))
        
        return mid
    
    def _trap_pattern(self, bars):
        pattern = []
        ticks = 384
        
        for bar in range(bars):
            offset = bar * ticks
            
            # 808 kick on 1 and 3.5
            pattern.append((36, 100, offset if bar == 0 else 0))
            pattern.append((36, 100, ticks // 3))
            
            # Snare on 2 and 4
            pattern.append((38, 90, ticks // 4))
            pattern.append((38, 90, ticks // 2))
            
            # Rapid hi-hats (16th notes)
            for i in range(16):
                pattern.append((42, 60, ticks // 32))
        
        return pattern
    
    def _boom_bap_pattern(self, bars):
        pattern = []
        ticks = 384
        
        for bar in range(bars):
            offset = bar * ticks
            
            # Heavy kick on 1 and 3
            pattern.append((36, 110, offset if bar == 0 else 0))
            pattern.append((36, 110, ticks // 2))
            
            # Snare on 2 and 4
            pattern.append((38, 100, ticks // 4))
            pattern.append((38, 100, ticks // 4))
            
            # Laid back hi-hats (8th notes)
            for i in range(8):
                pattern.append((42, 70, ticks // 16))
        
        return pattern

```python
#!/usr/bin/env python3
"""
Gemini API Wrapper - REST Edition
NO grpcio dependency, just pure HTTP requests
"""

import requests
import json
import yaml
from pathlib import Path

class GeminiWrapper:
    """Direct REST API calls to Gemini - bypasses grpcio"""
    
    def __init__(self, config_path="shared/config/config.yaml"):
        config = yaml.safe_load(Path(config_path).read_text())
        self.api_key = config['api_keys']['gemini']
        
        if not self.api_key:
            raise ValueError("No Gemini API key in config.yaml")
        
        self.base_url = "https://generativelanguage.googleapis.com/v1beta/models"
        self.model = "gemini-2.0-flash-exp"
    
    def generate(self, prompt, response_format='text', temperature=0.7):
        """Generate content via REST API"""
        
        url = f"{self.base_url}/{self.model}:generateContent"
        
        headers = {
            "Content-Type": "application/json"
        }
        
        payload = {
            "contents": [{
                "parts": [{
                    "text": prompt
                }]
            }],
            "generationConfig": {
                "temperature": temperature,
                "topK": 40,
                "topP": 0.95,
                "maxOutputTokens": 8192,
            }
        }
        
        # Add API key to URL
        response = requests.post(
            f"{url}?key={self.api_key}",
            headers=headers,
            json=payload,
            timeout=60
        )
        
        if response.status_code != 200:
            raise Exception(f"API error: {response.status_code} - {response.text}")
        
        result = response.json()
        
        # Extract text from response
        text = result['candidates'][0]['content']['parts'][0]['text']
        
        if response_format == 'json':
            try:
                # Try to parse as JSON
                return json.loads(text)
            except:
                # If not valid JSON, return raw
                return {"raw": text}
        
        return text
    
    def chat(self, messages, temperature=0.7):
        """Multi-turn conversation"""
        
        url = f"{self.base_url}/{self.model}:generateContent"
        
        # Convert messages to Gemini format
        contents = []
        for msg in messages:
            role = "user" if msg["role"] == "user" else "model"
            contents.append({
                "role": role,
                "parts": [{"text": msg["content"]}]
            })
        
        payload = {
            "contents": contents,
            "generationConfig": {
                "temperature": temperature,
                "maxOutputTokens": 8192,
            }
        }
        
        response = requests.post(
            f"{url}?key={self.api_key}",
            headers={"Content-Type": "application/json"},
            json=payload,
            timeout=60
        )
        
        if response.status_code != 200:
            raise Exception(f"API error: {response.status_code}")
        
        result = response.json()
        return result['candidates'][0]['content']['parts'][0]['text']

if __name__ == '__main__':
    # Test it
    wrapper = GeminiWrapper()
    result = wrapper.generate("Say 'Code City is online - no grpcio needed'")
    print(result)
```

```bash
#!/usr/bin/env python3
import requests, json, yaml
from pathlib import Path

class GeminiWrapper:
    def __init__(self, config_path="shared/config/config.yaml"):
        config = yaml.safe_load(Path(config_path).read_text())
        self.api_key = config['api_keys']['gemini']
        if not self.api_key:
            raise ValueError("No API key")
        self.base_url = "https://generativelanguage.googleapis.com/v1beta/models"
        self.model = "gemini-2.0-flash-exp"
    
    def generate(self, prompt, response_format='text'):
        url = f"{self.base_url}/{self.model}:generateContent?key={self.api_key}"
        payload = {
            "contents": [{"parts": [{"text": prompt}]}],
            "generationConfig": {"temperature": 0.7, "maxOutputTokens": 8192}
        }
        response = requests.post(url, json=payload, timeout=60)
        if response.status_code != 200:
            raise Exception(f"API error: {response.status_code}")
        text = response.json()['candidates'][0]['content']['parts'][0]['text']
        if response_format == 'json':
            try:
                return json.loads(text)
            except:
                return {"raw": text}
        return text

if __name__ == '__main__':
    w = GeminiWrapper()
    print(w.generate("Say 'Online'"))
EOFGEMINI

#!/usr/bin/env python3
from mido import MidiFile, MidiTrack, Message

class BeatGenerator:
    def __init__(self, bpm=140):
        self.bpm = bpm
    
    def create_beat(self, style='trap', bars=8):
        mid = MidiFile()
        track = MidiTrack()
        mid.tracks.append(track)
        
        pattern = []
        ticks = 384
        
        for bar in range(bars):
            offset = bar * ticks
            pattern.append((36, 100, offset if bar == 0 else 0))  # Kick
            pattern.append((38, 90, ticks//2))  # Snare
            for i in range(8):
                pattern.append((42, 60, ticks//16))  # Hi-hat
        
        for note, vel, time in pattern:
            track.append(Message('note_on', channel=9, note=note, velocity=vel, time=time))
            track.append(Message('note_off', channel=9, note=note, time=96))
        
        return mid

if __name__ == '__main__':

    gen.create_beat().save('/tmp/beat.mid')
```

```bash
#!/usr/bin/env python3
import requests, json, yaml
from pathlib import Path

class GeminiWrapper:
    def __init__(self, config_path="shared/config/config.yaml"):
        config = yaml.safe_load(Path(config_path).read_text())
        self.api_key = config['api_keys']['gemini']
        self.base_url = "https://generativelanguage.googleapis.com/v1beta/models"
        self.model = "gemini-2.0-flash-exp"
    
    def generate(self, prompt, response_format='text'):
        url = f"{self.base_url}/{self.model}:generateContent?key={self.api_key}"
        payload = {
            "contents": [{"parts": [{"text": prompt}]}],
            "generationConfig": {"temperature": 0.7, "maxOutputTokens": 8192}
        }
        response = requests.post(url, json=payload, timeout=60)
        text = response.json()['candidates'][0]['content']['parts'][0]['text']
        if response_format == 'json':
            try:
                return json.loads(text)
            except:
                return {"raw": text}
        return text
```

## SOLVING SPECIFIC PHONE VIDEO PROBLEMS

### **Problem: Shaky Footage**
```
Solutions:
1. Digital stabilization in post (CapCut has this)
2. Brace phone against wall/table when shooting
3. Use tripod for static shots, handheld for energy
4. Lean into shakiness for "found footage" feel
5. Cut every 3-5 seconds (motion sickness prevention)
```

### **Problem: Bad Low-Light Performance**
```
Solutions:
1. Shoot during day near windows
2. Use ring light (even cheap one helps)
3. Increase ISO in camera app (grainy but visible)
4. Embrace the grain (add MORE in post for aesthetic)
5. Shoot silhouettes if light is behind you (looks intentional)
```

### **Problem: Audio Echo/Reverb**
```
Solutions:
1. Get closer to mic (6-12 inches max)
2. Hang blankets on walls (cheapest sound treatment)
3. Shoot in smaller spaces (less echo)
4. Use noise reduction in post (Audacity, free)
5. Voiceover in car/closet (re-record bad takes)
```

### **Problem: Boring Visuals**
```
Solutions:
1. Shoot different angles of same thing
2. Add text overlays (highlight key points)
3. Use split-screen (you vs AI output)
4. Animate text/graphics (CapCut templates)
5. Stock footage (Pexels.com is free)
```

### **Problem: File Management Hell**
```
Solutions:
1. Name clips as you shoot (voice memo: "scene 3 take 2")
2. Transfer to cloud immediately (Google Drive, Dropbox)
3. Delete bad takes on-location (saves storage)
4. Use folders: RAW, WORKING, FINAL
5. External storage (USB-C drive, $30 for 256GB)
```

## COLOR GRADING (Make It Look "Cinematic")

### **The "Skin Deep" Look:**
```
CapCut Preset Recipe:
1. Contrast: +20
2. Shadows: -15 (crushed blacks)
3. Highlights: -10 (protect bright areas)
4. Saturation: -20 (desaturated, gritty)
5. Temperature: Slightly cool (blue tint)
6. Vignette: Light (darkens edges)

Result: High-contrast, gritty, "tech noir" aesthetic

Save as preset, apply to all clips (consistency is key).
```

### **Alternative Looks:**
- **Documentary:** Neutral, high detail, no color grading
- **Music Video:** High saturation, warm tones, lots of contrast
- **Tutorial:** Bright, clean, slightly overexposed (easy to see details)

**Pro Tip:** Steal a look you like. Screenshot Netflix show, match colors in your editor.

```python
# obelisk/vault/secrets.py
"""
Vault - Encrypted Secret Management
API keys, passwords, sensitive config - stored safely
"""

from cryptography.fernet import Fernet
import json
import os
from pathlib import Path

class Vault:
    """Secrets stay secret"""
    
    def __init__(self, master_password):
        self.key = self._derive_key(master_password)
        self.cipher = Fernet(self.key)
        self.vault_path = Path('shared/data/.vault')
        self.vault_path.touch(mode=0o600)  # Read/write for owner only
    
    def _derive_key(self, password):
        """Turn password into encryption key"""
        from cryptography.hazmat.primitives import hashes
        from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
        import base64
        
        kdf = PBKDF2HMAC(
            algorithm=hashes.SHA256(),
            length=32,
            salt=b'code_city_salt',  # In production, use random salt
            iterations=100000,
        )
        key = base64.urlsafe_b64encode(kdf.derive(password.encode()))
        return key
    
    def store_secret(self, name, value):
        """Encrypt and save secret"""
        secrets = self._load_vault()
        secrets[name] = value
        
        encrypted = self.cipher.encrypt(json.dumps(secrets).encode())
        self.vault_path.write_bytes(encrypted)
    
    def get_secret(self, name):
        """Retrieve decrypted secret"""
        secrets = self._load_vault()
        return secrets.get(name)
    
    def list_secrets(self):
        """Show secret names (not values)"""
        secrets = self._load_vault()
        return list(secrets.keys())
    
    def delete_secret(self, name):
        """Remove secret from vault"""
        secrets = self._load_vault()
        if name in secrets:
            del secrets[name]
            encrypted = self.cipher.encrypt(json.dumps(secrets).encode())
            self.vault_path.write_bytes(encrypted)
    
    def _load_vault(self):
        """Decrypt vault contents"""
        if self.vault_path.stat().st_size == 0:
            return {}
        
        encrypted = self.vault_path.read_bytes()
        decrypted = self.cipher.decrypt(encrypted)
        return json.loads(decrypted)
    
    def rotate_key(self, new_password):
        """Change master password"""
        secrets = self._load_vault()  # Decrypt with old key
        self.key = self._derive_key(new_password)
        self.cipher = Fernet(self.key)
        
        # Re-encrypt with new key
        encrypted = self.cipher.encrypt(json.dumps(secrets).encode())
        self.vault_path.write_bytes(encrypted)
    
    def export_backup(self, backup_path):
        """Encrypted backup (keep master password safe!)"""
        import shutil
        shutil.copy(self.vault_path, backup_path)
    
    def emergency_decrypt(self):
        """Print all secrets (use only in emergency)"""
        print("⚠️  EMERGENCY DECRYPT - SECRETS IN PLAIN TEXT ⚠️")
        secrets = self._load_vault()
        for name, value in secrets.items():
            print(f"{name}: {value}")
```

```python
"""
Beacon - Public Status Page & Uptime Monitoring
Let customers know if you're up or down
"""

from flask import Flask, render_template, jsonify
import requests
from datetime import datetime, timedelta
import json

app = Flask(__name__)

class Beacon:
    """Is it working? Let's find out."""
    
    def __init__(self):
        self.services = {}
        self.incidents = []
        self.uptime_data = {}
    
    def register_service(self, name, check_url, check_interval=60):
        """Add service to monitor"""
        self.services[name] = {
            'url': check_url,
            'interval': check_interval,
            'status': 'unknown',
            'last_check': None,
            'response_time': None
        }
    
    def check_service(self, service_name):
        """Ping service, record status"""
        service = self.services[service_name]
        
        try:
            start = datetime.now()
            response = requests.get(service['url'], timeout=10)
            elapsed = (datetime.now() - start).total_seconds()
            
            if response.status_code == 200:
                status = 'operational'
            else:
                status = 'degraded'
            
            self.services[service_name].update({
                'status': status,
                'last_check': datetime.now(),
                'response_time': elapsed
            })
            
            # Record uptime data
            self._record_uptime(service_name, status)
            
        except Exception as e:
            self.services[service_name]['status'] = 'down'
            self._create_incident(service_name, str(e))
    
    def _record_uptime(self, service_name, status):
        """Track historical uptime"""
        if service_name not in self.uptime_data:
            self.uptime_data[service_name] = []
        
        self.uptime_data[service_name].append({
            'timestamp': datetime.now().isoformat(),
            'status': status
        })
        
        # Keep last 7 days only
        cutoff = datetime.now() - timedelta(days=7)
        self.uptime_data[service_name] = [
            d for d in self.uptime_data[service_name]
            if datetime.fromisoformat(d['timestamp']) > cutoff
        ]
    
    def _create_incident(self, service_name, error):
        """Log incident"""
        incident = {
            'id': len(self.incidents) + 1,
            'service': service_name,
            'error': error,
            'started': datetime.now().isoformat(),
            'resolved': None,
            'status': 'investigating'
        }
        self.incidents.append(incident)
    
    def resolve_incident(self, incident_id, resolution):
        """Mark incident as resolved"""
        for incident in self.incidents:
            if incident['id'] == incident_id:
                incident['resolved'] = datetime.now().isoformat()
                incident['status'] = 'resolved'
                incident['resolution'] = resolution
    
    def calculate_uptime(self, service_name, days=7):
        """Percentage uptime over period"""
        data = self.uptime_data.get(service_name, [])
        if not data:
            return 100.0
        
        up_count = sum(1 for d in data if d['status'] == 'operational')
        total = len(data)
        
        return (up_count / total) * 100 if total > 0 else 100.0
    
    def generate_status_page(self):
        """HTML status page"""
        html = """
        <!DOCTYPE html>
        <html>
        <head>
            <title>Code City Status</title>
            <style>
                body { font-family: sans-serif; max-width: 800px; margin: 50px auto; }
                .service { padding: 15px; margin: 10px 0; border-radius: 5px; }
                .operational { background: #d4edda; color: #155724; }
                .degraded { background: #fff3cd; color: #856404; }
                .down { background: #f8d7da; color: #721c24; }
                .uptime { font-size: 24px; font-weight: bold; }
            </style>
        </head>
        <body>
            <h1>Code City System Status</h1>
            <p>Last updated: {timestamp}</p>
            
            {services_html}
            
            <h2>Recent Incidents</h2>
            {incidents_html}
        </body>
        </html>
        """
        
        # Generate services HTML
        services_html = ""
        for name, service in self.services.items():
            uptime = self.calculate_uptime(name)
            services_html += f"""
            <div class="service {service['status']}">
                <h3>{name}</h3>
                <p>Status: {service['status']}</p>
                <p>Response time: {service['response_time']:.2f}s</p>
                <p class="uptime">{uptime:.2f}% uptime (7 days)</p>
            </div>
            """
        
        # Generate incidents HTML
        incidents_html = "<ul>"
        for incident in sorted(self.incidents, key=lambda x: x['started'], reverse=True)[:5]:
            status_color = 'green' if incident['status'] == 'resolved' else 'red'
            incidents_html += f"""
            <li style="color: {status_color}">
                <strong>{incident['service']}</strong>: {incident['error']}
                <br>Started: {incident['started']}
                {f"<br>Resolved: {incident['resolved']}" if incident['resolved'] else ""}
            </li>
            """
        incidents_html += "</ul>"
        
        return html.format(
            timestamp=datetime.now().strftime('%Y-%m-%d %H:%M:%S'),
            services_html=services_html,
            incidents_html=incidents_html
        )
```
