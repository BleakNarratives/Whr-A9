```javascript
function addWindows(buildingMesh, buildingData) {
  const windowCount = Math.min(buildingData.windows, 50);
  
  for (let i = 0; i < windowCount; i++) {
    const windowGeometry = new THREE.BoxGeometry(0.5, 0.5, 0.1);
    const windowMaterial = new THREE.MeshBasicMaterial({ color: 0xffff00 });
    const windowMesh = new THREE.Mesh(windowGeometry, windowMaterial);
    
    // Random positions on building faces
    const face = Math.floor(Math.random() * 4);
    const offsetX = (Math.random() - 0.5) * buildingData.width;
    const offsetY = Math.random() * buildingData.height;
    
    // Position on different sides of building
    switch(face) {
      case 0: windowMesh.position.set(offsetX, offsetY, buildingData.depth/2); break;
      case 1: windowMesh.position.set(offsetX, offsetY, -buildingData.depth/2); break;
      // ... etc
    }
    
    buildingMesh.add(windowMesh); // Attach window to building
  }
}
```

```javascript
function addWindows(buildingMesh, buildingData) {
  const windowCount = Math.min(buildingData.windows, 50);
  
  for (let i = 0; i < windowCount; i++) {
    const windowGeometry = new THREE.BoxGeometry(0.5, 0.5, 0.1);
    const windowMaterial = new THREE.MeshBasicMaterial({ color: 0xffff00 });
    const windowMesh = new THREE.Mesh(windowGeometry, windowMaterial);
```

Real-Time Code Agents - HELL YES This Is The Money Maker!

Add this to your JavaScript for live code analysis:

```javascript
// AI Agent System
const codeAgents = [];

class CodeAgent {
    constructor(type, targetBuilding) {
        this.type = type; // 'debugger', 'tester', 'refactorer'
        this.target = targetBuilding;
        this.position = new THREE.Vector3();
        this.mesh = this.createAgentMesh();
        this.health = 100;
        this.speed = 0.5;
    }
    
    createAgentMesh() {
        let geometry, color;
        
        switch(this.type) {
            case 'debugger':
                geometry = new THREE.OctahedronGeometry(1);
                color = 0x00ff00; // Green
                break;
            case 'tester':
                geometry = new THREE.DodecahedronGeometry(1);
                color = 0xffff00; // Yellow
                break;
            case 'refactorer':
                geometry = new THREE.IcosahedronGeometry(1);
                color = 0xff00ff; // Pink
                break;
        }
        
        const material = new THREE.MeshBasicMaterial({ color });
        return new THREE.Mesh(geometry, material);
    }
    
    update() {
        // Move toward target building
        const direction = new THREE.Vector3()
            .subVectors(this.target.position, this.position)
            .normalize();
        
        this.position.add(direction.multiplyScalar(this.speed));
        this.mesh.position.copy(this.position);
        
        // If close to building, "fix" it
        if (this.position.distanceTo(this.target.position) < 5) {
            this.fixBuilding();
        }
    }
    
    fixBuilding() {
        const building = this.target.userData;
        
        switch(this.type) {
            case 'debugger':
                building.health += 2; // Fix bugs
                break;
            case 'tester':
                building.health += 1; // Prevent future issues
                break;
            case 'refactorer':
                building.health += 3; // Major improvements
                break;
        }
        
        // Cap health at 100
        building.health = Math.min(100, building.health);
        
        // Update building color based on health
        this.updateBuildingAppearance();
        
        // Agent takes damage
        this.health -= 10;
        if (this.health <= 0) {
            this.destroy();
        }
    }
    
    updateBuildingAppearance() {
        const building = this.target.userData;
        let color;
        
        if (building.health <= 20) {
            color = 0xff0000; // Red
        } else if (building.health <= 60) {
            color = 0xffa500; // Orange
        } else {
            color = parseInt(building.color.substring(1), 16);
        }
        
        this.target.material.color.setHex(color);
    }
    
    destroy() {
        scene.remove(this.mesh);
        codeAgents.splice(codeAgents.indexOf(this), 1);
    }
}

// Deploy agents when building is clicked
function deployAgent(type, buildingMesh) {
    const agent = new CodeAgent(type, buildingMesh);
    agent.position.set(
        camera.position.x + (Math.random() - 0.5) * 10,
        5,
        camera.position.z + (Math.random() - 0.5) * 10
    );
    
    scene.add(agent.mesh);
    codeAgents.push(agent);
    
    // Update UI to show active agents
    updateAgentCount();
}

// Update animation loop to handle agents
function animate() {
    requestAnimationFrame(animate);
    
    // Update all agents
    codeAgents.forEach(agent => agent.update());
    
    // Rest of your existing animation code...
    renderer.render(scene, camera);
}
```

        class CodeAgent {
            constructor(type, targetBuilding) {
                this.type = type;
                this.target = targetBuilding;
                this.position = new THREE.Vector3();
                this.mesh = this.createAgentMesh();
                this.health = 100;
                this.speed = 0.5;

            createAgentMesh() {
                let geometry, color;
                
                switch(this.type) {
                    case 'debugger':
                        geometry = new THREE.OctahedronGeometry(1);
                        color = 0x00ff00;
                        break;
                    case 'tester':
                        geometry = new THREE.DodecahedronGeometry(1);
                        color = 0xffff00;
                        break;
                    case 'refactorer':
                        geometry = new THREE.IcosahedronGeometry(1);
                        color = 0xff00ff;
                        break;
                }
                
                const material = new THREE.MeshBasicMaterial({ color });
                return new THREE.Mesh(geometry, material);
            }
            
            update() {
                const direction = new THREE.Vector3()
                    .subVectors(this.target.position, this.position)
                    .normalize();
                
                this.position.add(direction.multiplyScalar(this.speed));
                this.mesh.position.copy(this.position);
                
                if (this.position.distanceTo(this.target.position) < 5) {
                    this.fixBuilding();
                }
            }
            
            fixBuilding() {
                const building = this.target.userData;
                
                switch(this.type) {
                    case 'debugger': building.health += 2; break;
                    case 'tester': building.health += 1; break;
                    case 'refactorer': building.health += 3; break;

                building.health = Math.min(100, building.health);
                this.updateBuildingAppearance();
                
                this.health -= 10;
                if (this.health <= 0) {

                this.target.material.color.setHex(color);
            }
            
            destroy() {
                scene.remove(this.mesh);
                codeAgents.splice(codeAgents.indexOf(this), 1);
                updateAgentCount();
            }
        }
